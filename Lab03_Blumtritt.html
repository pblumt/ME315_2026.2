f_filtro_voos <- function(chunk, pos) {
  chunk %>%
    select(DESTINATION_AIRPORT, ARRIVAL_DELAY) %>%
    filter(!is.na(DESTINATION_AIRPORT), !is.na(ARRIVAL_DELAY)) %>%
    filter(!grepl("^1", DESTINATION_AIRPORT))
}
